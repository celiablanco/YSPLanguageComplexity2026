Մարդու իրավունքների համընդհանուր հռչակագիր 
Նախաբան 
Մինչդեռ բնածին արժանապատվության և հավասարի ու անօտարելիության ճանաչումը 
Մարդկային ընտանիքի բոլոր անդամների իրավունքները ազատության, արդարության հիմքն են 
և խաղաղություն աշխարհում,  
Մինչդեռ մարդու իրավունքների անտեսումն ու արհամարհանքը բերել են բարբարոսության 
գործողություններ, որոնք զայրացրել են մարդկության խիղճը և աշխարհի գալուստը 
որում մարդիկ պետք է օգտվեն խոսքի և հավատքի ազատությունից և ազատությունից 
վախից ու կարիքից հռչակվել է որպես ընդհանուրի բարձրագույն ձգտում 
մարդիկ,  
Մինչդեռ դա էական է, եթե մարդը ստիպված չլինի դիմելու, որպես վերջին հաշվով 
դիմել, ապստամբության ընդդեմ բռնակալության և ճնշումների, որոնք պետք է լինեն մարդու իրավունքները 
պաշտպանված է օրենքի գերակայությամբ,  
Մինչդեռ կարևոր է նպաստել միջև բարեկամական հարաբերությունների զարգացմանը 
ազգեր,  
Մինչդեռ Միավորված ազգերի կազմակերպության ժողովուրդները Կանոնադրության մեջ վերահաստատել են իրենց 
հավատ մարդու հիմնարար իրավունքների, մարդու արժանապատվության և արժեքի նկատմամբ 
և տղամարդկանց և կանանց հավասար իրավունքների մեջ և որոշել են նպաստել 
սոցիալական առաջընթաց և ավելի մեծ ազատության պայմաններում կյանքի ավելի լավ չափանիշներ,  
Մինչդեռ անդամ երկրները պարտավորվել են հասնել համագործակցության 
Միավորված ազգերի կազմակերպության հետ համընդհանուր հարգանքի և պահպանման խթանում 
մարդու իրավունքները և հիմնարար ազատությունները,  
Մինչդեռ այս իրավունքների և ազատությունների ընդհանուր ըմբռնումը պատկանում է 
մեծագույն նշանակություն այս խոստման լիարժեք իրականացման համար,  
Հիմա, հետևաբար,
Ընդհանուր ժողովը,  
Հռչակում է Մարդու իրավունքների այս համընդհանուր հռչակագիրը որպես ընդհանուր չափանիշ 
ձեռքբերում բոլոր ժողովուրդների և բոլոր ազգերի համար, որպեսզի յուրաքանչյուր անհատ և 
Հասարակության յուրաքանչյուր մարմին, մշտապես նկատի ունենալով այս Հռչակագիրը, պետք է ձգտի դրան 
 

 
ուսուցում և կրթություն՝ խթանելու այս իրավունքների և ազատությունների հարգանքը և 
առաջադեմ միջոցառումներ՝ ազգային և միջազգային, ապահովելու դրանց համընդհանուր և 
արդյունավետ ճանաչում և պահպանում, ինչպես անդամ պետությունների ժողովուրդների միջև 
իրենց և իրենց իրավասության տակ գտնվող տարածքների ժողովուրդների միջև:  
Հոդված I  
Բոլոր մարդիկ ծնվում են ազատ և հավասար արժանապատվության և իրավունքների մեջ: Նրանք են 
օժտված են բանականությամբ և խղճով և պետք է գործեն միմյանց նկատմամբ ա 
եղբայրության ոգի.  
Հոդված 2  
Յուրաքանչյուր ոք ունի սույն Հռչակագրում ամրագրված բոլոր իրավունքների և ազատությունների իրավունքը, 
առանց որևէ տեսակի տարբերության, օրինակ՝ ռասայական, գույնի, սեռի, լեզվի, կրոնի, 
քաղաքական կամ այլ կարծիք, ազգային կամ սոցիալական ծագում, ունեցվածք, ծնունդ կամ այլ կարգավիճակ:  
Ավելին, ոչ մի տարբերակում չպետք է դրվի քաղաքական, 
երկրի կամ տարածքի իրավասության կամ միջազգային կարգավիճակը, որտեղ անձը 
պատկանում է, անկախ նրանից, թե դա անկախ է, վստահության, ոչ ինքնակառավարման կամ որևէ այլ անձի տակ 
ինքնիշխանության սահմանափակում.  
Հոդված 3  
Յուրաքանչյուր ոք ունի կյանքի, ազատության և անձնական անվտանգության իրավունք։  
Հոդված 4  
Ոչ ոք չպետք է պահվի ստրկության կամ ստրկության մեջ. կլինի ստրկությունը և ստրկավաճառությունը
արգելված են իրենց բոլոր ձևերով:  
Հոդված 5  
Ոչ ոք չպետք է ենթարկվի խոշտանգումների կամ դաժան, անմարդկային կամ նվաստացնող վերաբերմունքի 
կամ պատիժ.  
 

 
Հոդված 6  
Յուրաքանչյուր ոք իրավունք ունի օրենքի առջև ամենուր անձ ճանաչվելու։  
Հոդված 7  
Բոլորը հավասար են օրենքի առաջ և առանց որևէ խտրականության հավասարվելու իրավունք ունեն 
օրենքի պաշտպանությունը։ Բոլորն ունեն հավասար պաշտպանության իրավունք որևէ մեկի դեմ 
խտրականություն՝ խախտելով սույն Հռչակագիրը և որևէ դրդման դեմ 
խտրականություն.  
Հոդված 8  
Յուրաքանչյուր ոք ունի իրավասու ազգային տրիբունալների կողմից իրավական պաշտպանության արդյունավետ միջոցի իրավունք 
Սահմանադրությամբ կամ օրենքով իրեն տրված հիմնարար իրավունքները ոտնահարող արարքների համար։  
Հոդված 9  
Ոչ ոք չպետք է ենթարկվի կամայական ձերբակալության, կալանավորման կամ աքսորի։  
Հոդված 10  
Յուրաքանչյուր ոք ունի անկախ անձի կողմից արդար և հրապարակային լսումների լիարժեք իրավահավասարության իրավունք 
և անկողմնակալ տրիբունալ՝ իր իրավունքների և պարտականությունների և որևէ մեկի որոշման հարցում 
նրա դեմ քրեական գործ է հարուցվել։  
Հոդված 11  
1. Յուրաքանչյուր ոք, ով մեղադրվում է քրեական հանցագործության համար, ունի ենթադրյալ իրավունք 
անմեղ է, քանի դեռ նրա մեղավորությունն ապացուցված չէ օրենքով սահմանված հրապարակային դատավարության ժամանակ, որում նա 
ունեցել է իր պաշտպանության համար անհրաժեշտ բոլոր երաշխիքները։  
2. Ոչ ոք չպետք է մեղավոր ճանաչվի որևէ քրեական հանցագործության համար որևէ արարքի կամ 
բացթողում, որը չի հանդիսանում քրեական հանցագործություն՝ ազգային կամ 
միջազգային իրավունքը, այն ժամանակ, երբ այն կատարվել է։ Ոչ էլ ավելի ծանր
տույժը պետք է նշանակվի, քան այն, որը կիրառելի էր պատժի պահին 
հանցագործություն է կատարվել.  
Հոդված 12  
Ոչ ոք չպետք է ենթարկվի կամայական միջամտության իր անձնական կյանքին, ընտանիքին, տանը 
կամ նամակագրություն, ոչ էլ նրա պատվի ու հեղինակության վրա հարձակումներ: Բոլորն էլ ունեն 
նման միջամտություններից կամ հարձակումներից օրենքի պաշտպանության իրավունքը։  
Հոդված 13  
1. Յուրաքանչյուր ոք ունի ազատ տեղաշարժվելու և բնակվելու իրավունք 
յուրաքանչյուր պետության սահմանները։  
2. Յուրաքանչյուր ոք իրավունք ունի լքել ցանկացած երկիր, ներառյալ իր երկիրը, և գնալ 
վերադառնալ իր երկիր.  
Հոդված 14  
1. Յուրաքանչյուր ոք ունի այլ երկրներում ապաստան փնտրելու և ստանալու իրավունք 
հալածանք։  
2. Այս իրավունքը չի կարող կիրառվել իրական հետապնդման դեպքում 
բխող ոչ քաղաքական հանցագործություններից կամ նպատակներին հակասող արարքներից և 
Միավորված ազգերի կազմակերպության սկզբունքները։  
Հոդված 15  
1. Յուրաքանչյուր ոք ունի ազգության իրավունք:  
2. Ոչ ոքի չի կարելի կամայականորեն զրկել իր քաղաքացիությունից կամ զրկել դրա իրավունքից 
փոխել իր ազգությունը.  
Հոդված 16  
 

 
1. Հասուն տարիքի տղամարդիկ և կանայք՝ առանց ռասայական, ազգության սահմանափակման 
կամ կրոն, իրավունք ունեն ամուսնանալու և ընտանիք կազմելու։ Նրանք իրավունք ունեն 
հավասար իրավունքներ ունենալ ամուսնության, ամուսնության ընթացքում և դրա լուծարման ժամանակ:  
2. Ամուսնությունը կնքվում է միայն նրա ազատ և լիակատար համաձայնությամբ 
մտադրություն ունեցող ամուսիններ.  
3. Ընտանիքը հասարակության բնական և հիմնարար խմբային միավորն է և է
հասարակության և պետության կողմից պաշտպանության իրավունք:  
Հոդված 17  
1. Յուրաքանչյուր ոք ունի սեփականություն ունենալու իրավունք ինչպես միայնակ, այնպես էլ նրա հետ միասին 
մյուսները։  
2. Ոչ ոքի չի կարելի կամայականորեն զրկել իր սեփականությունից։  
Հոդված 18  
Յուրաքանչյուր ոք ունի մտքի, խղճի և կրոնի ազատության իրավունք. այս իրավունքը 
ներառում է իր կրոնը կամ համոզմունքը փոխելու ազատությունը և ազատությունը՝ միայնակ կամ ներսում 
համայնքը ուրիշների հետ և հրապարակավ կամ մասնավոր կերպով՝ դրսևորելու իր կրոնը կամ հավատքը 
ուսուցում, պրակտիկա, պաշտամունք և հարգանք:  
Հոդված 19  
Յուրաքանչյուր ոք ունի կարծիքի և արտահայտվելու ազատության իրավունք. այս իրավունքը ներառում է 
առանց միջամտության կարծիքներ ունենալու և փնտրելու, ստանալու և տարածելու ազատություն 
տեղեկատվություն և գաղափարներ ցանկացած լրատվամիջոցի միջոցով և անկախ սահմաններից:  
Հոդված 20  
1. Յուրաքանչյուր ոք ունի խաղաղ հավաքների և միավորումների ազատության իրավունք:  
2. Ոչ ոքի չի կարելի ստիպել պատկանել միության:  
Հոդված 21  
 

 
1. Յուրաքանչյուր ոք ունի իր երկրի կառավարմանը մասնակցելու իրավունք, 
ուղղակիորեն կամ ազատորեն ընտրված ներկայացուցիչների միջոցով։  
2. Յուրաքանչյուր ոք ունի իր երկրում հանրային ծառայության հավասար հասանելիության իրավունք:  
3. Կառավարության իշխանության հիմքում ընկած է ժողովրդի կամքը. 
այս կամքը պետք է արտահայտվի պարբերական և իրական ընտրություններում, որոնք պետք է 
լինի համընդհանուր և հավասար ընտրական իրավունքով և անցկացվի գաղտնի քվեարկությամբ կամ քվեարկությամբ 
համարժեք ազատ քվեարկության ընթացակարգեր։  
Հոդված 22
Յուրաքանչյուր ոք, որպես հասարակության անդամ, ունի սոցիալական ապահովության իրավունք և ունի իրավունք 
իրականացնել ազգային ջանքերի և միջազգային համագործակցության միջոցով և 
յուրաքանչյուր պետության կազմակերպությանը և ռեսուրսներին համապատասխան, տնտեսական, 
նրա արժանապատվության և ազատ զարգացման համար անփոխարինելի սոցիալական և մշակութային իրավունքներ 
իր անձի մասին։  
Հոդված 23  
1. Յուրաքանչյուր ոք ունի աշխատելու, աշխատանքի ազատ ընտրության, արդար և 
աշխատանքի բարենպաստ պայմաններ և պաշտպանություն գործազրկությունից:  
2. Յուրաքանչյուր ոք, առանց որևէ խտրականության, ունի հավասար վարձատրության իրավունք 
աշխատանք.  
3. Յուրաքանչյուր ոք, ով աշխատում է, ունի արդար և բարենպաստ վարձատրության իրավունք 
ապահովելով իր և իր ընտանիքի համար մարդկային արժանապատվությունը վայելող գոյություն, 
և անհրաժեշտության դեպքում լրացվում է սոցիալական պաշտպանության այլ միջոցներով:  
4. Յուրաքանչյուր ոք ունի արհմիություններ ստեղծելու և դրանց պաշտպանության համար անդամակցելու իրավունք 
նրա շահերը.  
Հոդված 24  
Յուրաքանչյուր ոք ունի հանգստի և ժամանցի իրավունք, ներառյալ ողջամիտ սահմանափակումը 
աշխատանքային ժամերը և վճարովի պարբերական արձակուրդները.  
 

 
Հոդված 25  
1. Յուրաքանչյուր ոք ունի առողջությանը համարժեք կենսամակարդակի իրավունք և 
իր և իր ընտանիքի բարեկեցությունը, ներառյալ սնունդը, հագուստը, բնակարանը 
և բժշկական օգնություն և անհրաժեշտ սոցիալական ծառայություններ, և ապահովության իրավունք 
գործազրկության, հիվանդության, հաշմանդամության, այրիության, ծերության կամ 
ապրուստի այլ պակաս՝ իր կամքից անկախ հանգամանքներում:
2. Մայրությունն ու մանկությունը հատուկ խնամքի և օգնության իրավունք ունեն։ Բոլորը 
երեխաները, անկախ այն բանից՝ ծնված են ամուսնության մեջ կամ դրանից դուրս, պետք է օգտվեն նույն սոցիալականից 
պաշտպանություն։  
Հոդված 26  
1. Յուրաքանչյուր ոք ունի կրթության իրավունք։ Կրթությունը պետք է լինի անվճար, առնվազն Հայաստանում 
տարրական և հիմնարար փուլեր. Տարրական կրթությունը պետք է լինի 
պարտադիր. Կատարվի տեխնիկական և մասնագիտական կրթություն 
ընդհանուր առմամբ մատչելի և բարձրագույն կրթությունը պետք է հավասարապես հասանելի լինի բոլորի համար 
վաստակի հիման վրա։  
2. Կրթությունը պետք է ուղղված լինի մարդու լիարժեք զարգացմանը 
անհատականությունը և մարդու իրավունքների նկատմամբ հարգանքի ամրապնդումը և 
հիմնարար ազատություններ։ Այն պետք է նպաստի փոխըմբռնմանը, հանդուրժողականությանը և 
բարեկամությունը բոլոր ազգերի, ռասայական կամ կրոնական խմբերի միջև և պետք է շարունակվի 
Միավորված ազգերի կազմակերպության գործունեությունը խաղաղության պահպանման համար։  
3. Ծնողները նախապես իրավունք ունեն ընտրելու այն կրթությունը, որը պետք է լինի 
տրվել են իրենց երեխաներին։  
Հոդված 27  
1. Յուրաքանչյուր ոք ունի իր մշակութային կյանքին ազատորեն մասնակցելու իրավունք 
համայնքը, վայելել արվեստը և մասնակցել գիտական առաջընթացին և 
դրա առավելությունները.  
 

 
2. Յուրաքանչյուր ոք ունի բարոյական և նյութական շահերի պաշտպանության իրավունք 
ցանկացած գիտական, գրական կամ գեղարվեստական արտադրանքի արդյունք, որի հեղինակն է նա 
հեղինակ.  
Հոդված 28  
Յուրաքանչյուր ոք ունի սոցիալական և միջազգային կարգի իրավունք, որում իրավունքները և 
սույն Հռչակագրով ամրագրված ազատությունները կարող են լիովին իրացվել:  
Հոդված 29
1. Յուրաքանչյուր ոք ունի պարտականություններ այն համայնքի հանդեպ, որտեղ միայն ազատ և լիարժեք են 
հնարավոր է նրա անհատականության զարգացումը.  
2. Յուրաքանչյուր ոք իր իրավունքների և ազատությունների իրականացման ընթացքում ենթակա է միայն 
այնպիսի սահմանափակումների, որոնք օրենքով սահմանված են բացառապես նպատակներով 
ապահովելով այլոց իրավունքների և ազատությունների պատշաճ ճանաչում և հարգանք 
բարոյականության, հասարակական կարգի և արդարացի պահանջների բավարարման համար 
ընդհանուր բարեկեցությունը ժողովրդավարական հասարակության մեջ:  
3. Այս իրավունքներն ու ազատությունները ոչ մի դեպքում չեն կարող իրականացվել սույն օրենքին հակասող 
Միավորված ազգերի կազմակերպության նպատակներն ու սկզբունքները։  
Հոդված 30  
Սույն Հռչակագրում ոչ մի բան չի կարող մեկնաբանվել որպես որևէ պետության, խմբի կամ 
անձը ցանկացած գործունեությամբ զբաղվելու կամ դրան ուղղված որևէ գործողություն կատարելու իրավունք ունի 
սույն օրենքով սահմանված իրավունքներից և ազատություններից որևէ մեկի ոչնչացում:
