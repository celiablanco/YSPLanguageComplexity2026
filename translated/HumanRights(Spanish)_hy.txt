Մարդու իրավունքների համընդհանուր հռչակագիր 
Ընդունվել և հռչակվել է Գլխավոր ասամբլեայի կողմից իր 217 Ա (III) որոշմամբ 
10 դեկտեմբերի, 1948 թ  
Նախաբան  
Նկատի ունենալով, որ աշխարհում ազատությունը, արդարությունն ու խաղաղությունը հիմնված են 
ներքին արժանապատվության և հավասար ու անօտարելի իրավունքների ճանաչում 
մարդկային ընտանիքի բոլոր անդամների,  
Նկատի ունենալով, որ անտեղյակությունն ու իրավունքների նկատմամբ արհամարհանքը 
մարդիկ բարբարոսության գործողություններ են կատարել, որոնք վիրավորական են մարդկանց խղճի համար 
մարդասիրություն; և դա հռչակվել է որպես բարձրագույն ձգտում 
մարդ, մի աշխարհի գալուստը, որտեղից ազատված մարդ արարածը 
վախ և թշվառություն, վայելեք խոսքի ազատությունը և ազատությունը 
համոզմունքները,  
Էական համարելով, որ մարդու իրավունքները պաշտպանվեն ա 
Օրենքի ռեժիմ, որպեսզի մարդը չպարտադրվի գերագույնին 
բռնակալության և ճնշումների դեմ ապստամբության ռեսուրս,  
Էական համարելով նաև բարեկամական հարաբերությունների զարգացմանը նպաստելը 
ազգերի մեջ,  
Նկատի ունենալով, որ Միավորված ազգերի կազմակերպության ժողովուրդները վերահաստատել են 
Նշեք ձեր հավատքը մարդու հիմնարար իրավունքների, արժանապատվության և արժեքի նկատմամբ 
մարդու անձի և տղամարդկանց և կանանց հավասար իրավունքները. և այն 
հայտարարել են սոցիալական առաջընթացը խթանելու և կենսամակարդակի բարձրացման իրենց վճռականությունը 
ազատության ավելի լայն հայեցակարգի շրջանակներում,
Նկատի ունենալով, որ անդամ պետությունները պարտավորվել են ապահովել, որ 
համագործակցությունը ՄԱԿ-ի հետ, համընդհանուր հարգանք և 
արդյունավետ է մարդու հիմնարար իրավունքների և ազատությունների համար, և  
Նկատի ունենալով, որ այս իրավունքների և ազատությունների ընդհանուր պատկերացումն է 
մեծագույն նշանակություն նշված պարտավորության լիարժեք կատարման համար,  
 

 
Ընդհանուր ժողովը  
Հռչակում է Մարդու իրավունքների ներկայիս համընդհանուր հռչակագիրը որպես իդեալ 
ընդհանուր նպատակ, որին պետք է ձգտեն բոլոր ժողովուրդներն ու ազգերը, որպեսզի 
թե՛ անհատներին, թե՛ հաստատություններին, մշտապես ոգեշնչվելով դրանից, 
ուսուցման և կրթության միջոցով նպաստել այս իրավունքների հարգմանը 
եւ ազատություններ, եւ ապահովել, առաջադիմական միջոցառումների միջոցով ազգային ու 
միջազգային իրավունքը, դրա համընդհանուր և արդյունավետ ճանաչումն ու կիրառումը, երկուսն էլ 
անդամ պետությունների, ինչպես նաև տեղակայված տարածքների ժողովուրդների շարքում 
իր իրավասության տակ։  
Հոդված 1  
Բոլոր մարդիկ ծնվում են ազատ և հավասար արժանապատվությամբ և իրավունքներով և, 
օժտված լինելով բանականությամբ և խղճով, նրանք պետք է եղբայրաբար վարվեն 
միմյանց հետ։  
Հոդված 2  
Յուրաքանչյուր ոք ունի սույն Հռչակագրով հռչակված իրավունքներն ու ազատությունները, 
առանց ռասայի, գույնի, սեռի, լեզվի, կրոնի, քաղաքական կարծիքի կամ տարբերության 
ցանկացած այլ բնույթ, ազգային կամ սոցիալական ծագում, տնտեսական դիրք, ծնունդ կամ 
ցանկացած այլ պայման:
Ավելին, որևէ տարբերակում չի արվելու՝ ելնելով քաղաքական, իրավական կամ քաղաքական կարգավիճակից։ 
այն երկրի կամ տարածքի միջազգային, որի իրավասության տակ է գտնվում անձը, 
անկախ երկիր է, թե տակ գտնվող տարածք 
հավատարմագրային կառավարում, որը ինքնավար չէ կամ ենթակա չէ որևէ այլ սահմանափակման 
ինքնիշխանություն։  
Հոդված 3  
Յուրաքանչյուր անհատ ունի կյանքի, ազատության և իր կյանքի անվտանգության իրավունք: 
մարդ.  
 

 
Հոդված 4  
Ոչ ոք չի ենթարկվի ստրկության կամ ստրկության. ստրկություն և թրաֆիքինգի 
Ստրուկներն արգելված են իրենց բոլոր ձևերով:  
Հոդված 5  
Ոչ ոք չպետք է ենթարկվի խոշտանգումների կամ դաժան, անմարդկային կամ դաժան վերաբերմունքի կամ պատժի։ 
ստորացուցիչ  
Հոդված 6  
Յուրաքանչյուր մարդ իրավունք ունի, ամենուր, ճանաչելու իրենց 
իրավաբանական անձ:  
Հոդված 7  
Բոլորը հավասար են օրենքի առջև և ունեն, առանց խտրականության, հավասար պաշտպանության իրավունք 
օրենքի։ Յուրաքանչյուր ոք ունի հավասար պաշտպանության իրավունք բոլոր խտրականությունից 
խախտում է սույն Հռչակագիրը և դեմ է նման խտրականության ցանկացած սադրանքի:  
Հոդված 8  
Յուրաքանչյուր ոք ունի իրավական պաշտպանության արդյունավետ միջոց ազգային դատարանների առջև։ 
իրավասու մարմիններին՝ պաշտպանելու նրանց իրավունքները ոտնահարող գործողություններից 
Սահմանադրությամբ կամ օրենքով ճանաչված հիմնարար իրավունքները։  
Հոդված 9  
Ոչ ոք չի կարող կամայականորեն կալանավորվել, բանտարկվել կամ աքսորվել։  
Հոդված 10  
Յուրաքանչյուր ոք իրավունք ունի լիակատար հավասարության պայմաններում լսելի լինելու
հրապարակայնորեն և արդարացիորեն անկախ և անաչառ դատարանի կողմից, 
 

 
նրանց իրավունքների և պարտականությունների որոշումը կամ որևէ մեկի քննությունը 
նրան քրեական գործով մեղադրանք առաջադրել։  
Հոդված 11  
1. Հանցագործության մեջ մեղադրվող յուրաքանչյուր անձ ունի իր ինքնությունը ենթադրյալ լինելու իրավունք 
անմեղությունը, քանի դեռ նրա մեղավորությունն ապացուցված չէ՝ օրենքով և ք 
հրապարակային դատավարություն, որում ապահովված են բոլոր երաշխիքները 
դրա պաշտպանության համար անհրաժեշտ:  
2. Ոչ ոք չի դատապարտվի այն արարքների կամ անգործության համար, որոնք ժամանակին 
կատարվածը հանցավոր չէ ազգային կամ միջազգային իրավունքի համաձայն: 
Չի կիրառվի նաև ավելի լուրջ տույժ, քան կիրառվել է տվյալ պահին 
հանցագործության կատարումը։  
Հոդված 12  
Ոչ ոք կամայական միջամտության չի ենթարկվի իր անձնական կյանքին, իր ընտանիքին, իր 
տունը կամ նրա նամակագրությունը, կամ հարձակումներ նրա պատվի կամ հեղինակության վրա: 
Յուրաքանչյուր ոք ունի օրենքով պաշտպանվելու իրավունք նման միջամտությունից կամ 
հարձակումներ.  
Հոդված 13  
1. Յուրաքանչյուր ոք ունի ազատ տեղաշարժվելու և իր բնակության վայր ընտրելու իրավունք 
պետության տարածքում։  
2. Յուրաքանչյուր ոք իրավունք ունի լքել ցանկացած երկիր, ներառյալ իր երկիրը, և 
վերադառնալ իր երկիր։  
Հոդված 14  
1. In case of persecution, everyone has the right to seek asylum, and to 
Վայելեք այն ցանկացած երկրում:  
 

 
2. Այս իրավունքը չի կարող վկայակոչվել իրապես դատական ակտի դեմ 
առաջացած ընդհանուր հանցագործություններից կամ նպատակներին հակասող արարքներով և
Միավորված ազգերի կազմակերպության սկզբունքները։  
Հոդված 15  
1. Յուրաքանչյուր ոք ունի ազգության իրավունք:  
2. Ոչ ոք չի կարող կամայականորեն զրկվել իր քաղաքացիությունից կամ իրավունքից 
change nationality.  
Հոդված 16  
1. Տղամարդիկ և կանայք, սկսած ամուսնության տարիքից, իրավունք ունեն՝ առանց 
ցանկացած սահմանափակում ռասայական, ազգության կամ կրոնի հիման վրա ամուսնանալու համար 
and found a family; և կօգտվեն հավասար իրավունքներից 
ամուսնությունը, ամուսնության ընթացքում և ամուսնության լուծարման դեպքում.  
2. Միայն ապագա ամուսինների ազատ և լիակատար համաձայնությամբ կարող է 
get married.  
3. Ընտանիքը հասարակության բնական և հիմնարար տարրն է և ունի 
հասարակության և պետության պաշտպանության իրավունքը։  
Հոդված 17  
1. Յուրաքանչյուր ոք ունի սեփականության իրավունք՝ անհատապես և հավաքականորեն։  
2. Ոչ ոք կամայականորեն չի զրկվի իր սեփականությունից.  
Հոդված 18  
Յուրաքանչյուր ոք ունի մտքի, խղճի և ազատության իրավունք 
religion; Այս իրավունքը ներառում է կրոնը կամ համոզմունքը փոխելու ազատությունը, ինչպես նաև 
որպես կրոնի կամ համոզմունքների դրսևորման ազատություն՝ անհատապես և 
հավաքականորեն, ինչպես հանրային, այնպես էլ մասնավոր, դասավանդման, պրակտիկայի միջոցով, 
worship and observance.  
 

 
Հոդված 19  
Յուրաքանչյուր անհատ ունի կարծիքի և արտահայտվելու ազատության իրավունք. սա 
իրավունքը ներառում է սեփական կարծիքի պատճառով չանհանգստանալը, հետաքննության իրավունքը և
ստանալ տեղեկատվություն և կարծիքներ և տարածել դրանք՝ առանց սահմանների սահմանափակման, 
ցանկացած արտահայտչական միջոցներով.  
Հոդված 20  
1. Յուրաքանչյուր ոք ունի հավաքների և միավորումների ազատության իրավունք 
խաղաղ.  
2. Ոչ ոքի չի կարելի ստիպել պատկանել ասոցիացիային:  
Հոդված 21  
1. Յուրաքանչյուր ոք ունի իր երկրի կառավարմանը մասնակցելու իրավունք, 
ուղղակիորեն կամ ազատորեն ընտրված ներկայացուցիչների միջոցով։  
2. Յուրաքանչյուր անձ ունի հավասարության պայմաններում հասանելիության իրավունք 
իրենց երկրի հանրային գործառույթները։  
3. Ժողովրդի կամքը հանրային իշխանության հեղինակության հիմքն է. սա 
կամքը կարտահայտվի վավերական ընտրությունների միջոցով, որոնք պետք է լինեն 
անցկացվի պարբերաբար, համընդհանուր և հավասար ընտրական իրավունքով և քվեարկությամբ 
գաղտնի կամ այլ համարժեք ընթացակարգ, որը երաշխավորում է նրա ազատությունը 
քվեարկել.  
Հոդված 22  
Յուրաքանչյուր մարդ, որպես հասարակության անդամ, ունի անվտանգության իրավունք 
սոցիալական և ազգային ջանքերի և միջազգային համագործակցության միջոցով ձեռք բերել, 
taking into account the organization and resources of each State, the satisfaction 
տնտեսական, սոցիալական և մշակութային իրավունքների, որոնք էական նշանակություն ունեն իրենց արժանապատվության համար 
և նրանց անձի ազատ զարգացումը:  
 

 
Հոդված 23  
1. Յուրաքանչյուր ոք ունի աշխատելու, աշխատանքի ազատ ընտրության իրավունք 
equitable and satisfactory conditions of work and protection against 
գործազրկություն. 
2. Յուրաքանչյուր ոք ունի հավասար վարձատրության իրավունք, առանց որևէ խտրականության
Ես նույնն եմ աշխատում։  
3. Յուրաքանչյուր անձ, ով աշխատում է, ունի արդար վարձատրության և 
գոհացուցիչ, որն ապահովում է նրան, ինչպես նաև իր ընտանիքին գոյությունը 
մարդկային արժանապատվությանը համապատասխան և դա կավարտվի, եթե 
necessary, by any other means of social protection.  
4. Յուրաքանչյուր ոք ունի արհմիություններ ստեղծելու և միություններին անդամակցելու իրավունք 
պաշտպանել իրենց շահերը.  
Հոդված 24  
Յուրաքանչյուր մարդ իրավունք ունի հանգստանալու, ազատ ժամանակ վայելելու, ա 
աշխատանքի և պարբերական արձակուրդների տևողության ողջամիտ սահմանափակում 
վճարված.  
Հոդված 25  
1. Յուրաքանչյուր ոք ունի համապատասխան կենսամակարդակի իրավունք, որն ապահովում է. 
ինչպես նաև ձեր ընտանիքը, առողջությունն ու բարեկեցությունը և հատկապես 
սնունդ, հագուստ, բնակարան, բժշկական օգնություն և ծառայություններ 
անհրաժեշտ սոցիալական Դուք նաև իրավունք ունեք ապահովագրության դեպքում 
գործազրկություն, հիվանդություն, հաշմանդամություն, այրիություն, ծերություն և կորստի այլ դեպքեր 
իրենց ապրուստի միջոցները՝ պայմանավորված իրենցից անկախ հանգամանքներով 
կամք.  
2. Մայրությունն ու մանկությունն ունեն խնամքի և օգնության իրավունք 
հատուկներ. Բոլոր երեխաները՝ անկախ ամուսնությունից ծնված, թե արտաամուսնական կապից 
ամուսնությունը, ունեն հավասար սոցիալական պաշտպանության իրավունք:  
 

 
Հոդված 26  
1. Յուրաքանչյուր ոք ունի կրթության իրավունք։ Կրթությունը պետք է լինի 
անվճար, գոնե տարրական կրթության մասով և 
հիմնարար. Տարրական կրթությունը պարտադիր է լինելու. Հրահանգը
տեխնիկական և մասնագիտական ​​պետք է ընդհանրացվեն. ուսումնասիրությունների հասանելիություն 
վերադասները բոլորի համար նույնն են լինելու՝ կախված իրենց արժանիքներից:  
2. Կրթությունը նպատակաուղղված կլինի անձի լիարժեք զարգացմանը 
մարդու իրավունքները և մարդու իրավունքների նկատմամբ հարգանքի ամրապնդումը և 
հիմնարար ազատություններ; կնպաստի փոխըմբռնմանը, հանդուրժողականությանը և 
բարեկամություն բոլոր ազգերի և բոլոր էթնիկ կամ կրոնական խմբերի միջև. և 
նպաստել Միավորված ազգերի կազմակերպության գործունեության զարգացմանը 
խաղաղապահ.  
3. Կրթության տեսակը ընտրելու արտոնյալ իրավունք կունենան ծնողները 
որը կտրվի իրենց երեխաներին։  
Հոդված 27  
1. Յուրաքանչյուր ոք ունի մշակութային կյանքին ազատորեն մասնակցելու իրավունք 
համայնքի, վայելել արվեստը և մասնակցել առաջընթացին 
գիտական և դրանից բխող օգուտները:  
2. Յուրաքանչյուր ոք ունի բարոյական շահերի պաշտպանության իրավունք և 
գիտական արտադրանքի շնորհիվ դրան համապատասխանող նյութեր, 
գրական կամ գեղարվեստական ստեղծագործություններ, որոնց հեղինակն է նա։  
Հոդված 28  
Յուրաքանչյուր ոք ունի սոցիալական և միջազգային կարգեր հաստատելու իրավունք 
որում ամրագրված են սույն Հռչակագրով հռչակված իրավունքներն ու ազատությունները 
լիովին արդյունավետ:  
Հոդված 29  
 

 
1. Յուրաքանչյուր մարդ պարտավորություններ ունի համայնքի նկատմամբ, քանի որ միայն ք 
նա կարող է ազատ և լիովին զարգացնել իր անհատականությունը:  
2. Իրենց իրավունքների իրականացման և իրենց ազատություններից օգտվելիս բոլորը
person will only be subject to the limitations established by law 
with the sole purpose of ensuring the recognition and respect of the 
rights and freedoms of others, and to satisfy the just demands 
of morality, public order and general well-being in a society 
democratic.  
3. These rights and freedoms may under no circumstances be exercised in 
opposition to the purposes and principles of the United Nations.
